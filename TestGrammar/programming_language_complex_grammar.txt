# 带if-else语句的编程语言文法
# 特点：包含变量声明、赋值、条件语句、循环、函数、数组等编程语言结构
# 复杂度：高，55个产生式，支持多种数据类型和控制结构
# 分析器兼容性：SLR(1), LR(1)
# 典型输入示例：
#   - 变量声明: int id ; | bool id ; | float id [ number ] ;
#   - 数组赋值: id [ number ] = number ;
#   - 函数调用: id ( number , id ) ;
#   - for循环: for ( id = number ; number > id ; id = id + number ) { id = number ; }
#   - if-else: if ( id > number ) { id = number ; } else { id = false ; }
#   - 嵌套结构: if ( id == true ) { id ( number ) ; } else { id [ number ] = false ; }

# 程序入口
Program -> StmtList

# 语句列表
StmtList -> Stmt
StmtList -> Stmt StmtList

# 语句类型
Stmt -> VarDecl
Stmt -> Assignment
Stmt -> IfStmt
Stmt -> WhileStmt
Stmt -> ForStmt
Stmt -> Block
Stmt -> ReturnStmt

# 变量声明
VarDecl -> Type id ;
VarDecl -> Type id = Expr ;
VarDecl -> Type id [ number ] ;

# 数据类型
Type -> int
Type -> bool
Type -> float

# 赋值语句
Assignment -> id = Expr ;
Assignment -> id [ Expr ] = Expr ;

# for循环中的简单赋值（不需要分号）
SimpleAssignment -> id = Expr
SimpleAssignment -> id [ Expr ] = Expr

# 条件语句
IfStmt -> if ( Expr ) Block
IfStmt -> if ( Expr ) Block else Block
IfStmt -> if ( Expr ) Block else IfStmt

# 循环语句
WhileStmt -> while ( Expr ) Block
ForStmt -> for ( Assignment Expr ; SimpleAssignment ) Block

# 函数调用
FuncCall -> id ( ArgList )
FuncCall -> id ( )

# 参数列表
ArgList -> Expr
ArgList -> Expr , ArgList

# 表达式语句和返回语句
ReturnStmt -> return Expr ;
ReturnStmt -> return ;

# 代码块
Block -> { StmtList }
Block -> { }

# 表达式
Expr -> CompareExpr

# 比较表达式
CompareExpr -> ArithExpr
CompareExpr -> ArithExpr < ArithExpr
CompareExpr -> ArithExpr > ArithExpr
CompareExpr -> ArithExpr == ArithExpr
CompareExpr -> ArithExpr != ArithExpr

# 算术表达式
ArithExpr -> Term
ArithExpr -> ArithExpr + Term
ArithExpr -> ArithExpr - Term

# 项
Term -> Factor
Term -> Term * Factor
Term -> Term / Factor

# 因子
Factor -> Primary
Factor -> ! Factor
Factor -> - Factor

# 基本表达式
Primary -> id
Primary -> number
Primary -> true
Primary -> false
Primary -> ( Expr )
Primary -> FuncCall
Primary -> id [ Expr ]
